2. A Regra do "Código Burro Primeiro":
O seu erro atual é tentar escrever o código perfeito, limpo e tipado logo na primeira linha.
- Escreva o código mais feio, estático e "burro" possível, apenas para funcionar.
- Precisa buscar dados de um usuário? Não crie a rota, o controller e o banco agora. Crie uma variável const usuario = { nome: "Walao" } no topo do arquivo e use ela.

1. O Serviço (src/services/PokemonService.ts):
Este arquivo é responsável por buscar os dados da API (sua etapa 6 e 7) e guardar a lista na memória.

// Define o formato do Pokémon que vamos salvar na nossa lista
interface Pokemon {
  id: number;
  name: string;
  types: string[];
}

export class PokemonService {
  // Array na memória para guardar os Pokémons (sua etapa 8)
  private pokemons: Pokemon[] = [];

  async fetchPokemonFromAPI(name: string): Promise<void> {
    // 1. Faz a requisição para a PokeAPI
    const response = await fetch(`https://pokeapi.co/api/v2/pokemon/${name}`);
    
    if (!response.ok) {
      throw new Error(`Pokémon ${name} não encontrado.`);
    }

    // 2. Converte a resposta em JSON
    const data = await response.json();

    // 3. Mapeia os dados que precisamos (ID, Nome e Tipos)
    const novoPokemon: Pokemon = {
      id: data.id,
      name: data.name,
      types: data.types.map((t: any) => t.type.name)
    };

    // 4. Salva na nossa lista interna
    this.pokemons.push(novoPokemon);
    console.log(`Pokémon ${name} encontrado na PokeAPI e salvo!`);
  }

  // Método para o Controller buscar a lista (sua etapa 8)
  getPokemons(): Pokemon[] {
    return this.pokemons;
  }
}


2. O Controller (src/controllers/PokemonController.ts):
Este arquivo serve como o "gerente". Ele recebe a ordem do main.ts e aciona o Service.

import { PokemonService } from "../services/PokemonService";

export class PokemonController {
  // O Controller precisa do Service para funcionar
  private pokemonService = new PokemonService();

  async fetchPokemon(name: string): Promise<void> {
    try {
      // Repassa a ordem para o serviço buscar o Pokémon
      await this.pokemonService.fetchPokemonFromAPI(name);
    } catch (error: any) {
      console.error(`Erro no Controller: ${error.message}`);
    }
  }

  listPokemons(): void {
    // Busca a lista salva no serviço
    const lista = this.pokemonService.getPokemons();
    
    console.log("\n=== Lista de Pokemon ===");
    lista.forEach(pokemon => {
      // Junta os tipos com uma vírgula (ex: "electric, steel")
      const tipos = pokemon.types.join(", ");
      console.log(`ID: ${pokemon.id} , Nome: ${pokemon.name} , Tipo: ${tipos}`);
    });
  }
}
